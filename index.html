const BASE = "https://speed.cloudflare.com"; async function testPing() { const results = []; for (let i = 0; i < 6; i++) { const start = performance.now(); await fetch( `${BASE}/__down?bytes=0&cache=${Math.random()}`, { cache: "no-store" } ); const end = performance.now(); results.push(end - start); } results.sort((a, b) => a - b); return results[Math.floor(results.length / 2)]; } async function testDownload() { const bytes = 25_000_000; const start = performance.now(); const response = await fetch( `${BASE}/__down?bytes=${bytes}&cache=${Math.random()}`, { cache: "no-store" } ); if (!response.ok) { throw new Error("Download test failed"); } const reader = response.body.getReader(); let received = 0; while (true) { const { done, value } = await reader.read(); if (done) break; received += value.length; const seconds = (performance.now() - start) / 1000; const mbps = (received * 8) / seconds / 1_000_000; console.log("Download live:", mbps); } const seconds = (performance.now() - start) / 1000; return ( received * 8 / seconds / 1_000_000 ); } async function testUpload() { const size = 10_000_000; const data = new Uint8Array(size); const start = performance.now(); const response = await fetch( `${BASE}/__up?cache=${Math.random()}`, { method: "POST", body: data, cache: "no-store" } ); if (!response.ok) { throw new Error("Upload test failed"); } const seconds = (performance.now() - start) / 1000; return ( size * 8 / seconds / 1_000_000 ); } async function runRealSpeedTest() { try { const ping = await testPing(); document.getElementById("ping").textContent = Math.round(ping); const download = await testDownload(); document.getElementById("download").textContent = download.toFixed(1); const upload = await testUpload(); document.getElementById("upload").textContent = upload.toFixed(1); } catch (error) { console.error(error); } }
